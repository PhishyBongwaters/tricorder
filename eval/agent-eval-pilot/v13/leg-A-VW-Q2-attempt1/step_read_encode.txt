106:     match jsonwebtoken::encode(&JWT_HEADER, claims, PRIVATE_RSA_KEY.wait()) {

109:     }

112: pub fn decode_jwt<T: DeserializeOwned>(token: &str, issuer: String) -> Result<T, Error> {
113:     let mut validation = jsonwebtoken::Validation::new(JWT_ALGORITHM);
114:     validation.leeway = 30; // 30 seconds
115:     validation.validate_exp = true;
116:     validation.validate_nbf = true;
117:     validation.set_issuer(&[issuer]);
118: 
119:     let token = token.replace(char::is_whitespace, "");
120:     match jsonwebtoken::decode(&token, PUBLIC_RSA_KEY.wait(), &validation) {
121:         Ok(d) => Ok(d.claims),
122:         Err(err) => match *err.kind() {
123:             ErrorKind::InvalidToken => err!("Token is invalid"),
124:             ErrorKind::InvalidIssuer => err!("Issuer is invalid"),
125:             ErrorKind::ExpiredSignature => err!("Token has expired"),
126:             _ => err!(format!("Error decoding JWT: {:?}", err)),
127:         },
128:     }
129: }
