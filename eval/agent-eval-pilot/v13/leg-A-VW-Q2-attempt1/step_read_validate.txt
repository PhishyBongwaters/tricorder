229: fn validate_token(token: &str) -> bool {
230:     match CONFIG.admin_token().as_ref() {
231:         None => false,
232:         Some(t) if t.starts_with("$argon2") => {
233:             use argon2::password_hash::PasswordVerifier;
234:             match argon2::password_hash::PasswordHash::new(t) {
235:                 Ok(h) => {
236:                     // NOTE: hash params from `ADMIN_TOKEN` are used instead of what is configured in the `Argon2` instance.
237:                     argon2::Argon2::default().verify_password(token.trim().as_ref(), &h).is_ok()
238:                 }
239:                 Err(e) => {
240:                     error!("The configured Argon2 PHC in `ADMIN_TOKEN` is invalid: {e}");
241:                     false
242:                 }
243:             }
244:         }
245:         Some(t) => crate::crypto::ct_eq(t.trim(), token.trim()),
246:     }
247: }
