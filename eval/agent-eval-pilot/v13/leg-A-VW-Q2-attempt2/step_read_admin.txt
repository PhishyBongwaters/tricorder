184: #[post("/", format = "application/x-www-form-urlencoded", data = "<data>")]
185: fn post_admin_login(
186:     data: Form<LoginForm>,
187:     cookies: &CookieJar<'_>,
188:     ip: ClientIp,
189:     secure: Secure,
190: ) -> Result<Redirect, AdminResponse> {
191:     let data = data.into_inner();
192:     let redirect = data.redirect;
193: 
194:     if crate::ratelimit::check_limit_admin(&ip.ip).is_err() {
195:         return Err(AdminResponse::TooManyRequests(render_admin_login(
196:             Some("Too many requests, try again later."),
197:             redirect.as_deref(),
198:         )));
199:     }
200: 
201:     // If the token is invalid, redirect to login page
202:     if validate_token(&data.token) {
203:         // If the token received is valid, generate JWT and save it as a cookie
204:         let claims = generate_admin_claims();
205:         let jwt = encode_jwt(&claims);
206: 
207:         let cookie = Cookie::build((COOKIE_NAME, jwt))
208:             .path(admin_path())
209:             .max_age(time::Duration::minutes(CONFIG.admin_session_lifetime()))
210:             .same_site(SameSite::Strict)
211:             .http_only(true)
212:             .secure(secure.https);
213: 
214:         cookies.add(cookie);
215:         if let Some(redirect) = redirect {
216:             Ok(Redirect::to(format!("{}{redirect}", admin_path())))
217:         } else {
218:             Err(AdminResponse::Ok(render_admin_page()))
219:         }
220:     } else {
221:         error!("Invalid admin token. IP: {}", ip.ip);
222:         Err(AdminResponse::Unauthorized(render_admin_login(
223:             Some("Invalid admin token, please try again."),
224:             redirect.as_deref(),
225:         )))
226:     }
227: }

229: fn validate_token(token: &str) -> bool {
230:     match CONFIG.admin_token().as_ref() {
231:         None => false,
232:         Some(t) if t.starts_with("$argon2") => {
233:             use argon2::password_hash::PasswordVerifier;
234:             match argon2::password_hash::PasswordHash::new(t) {
235:                 Ok(h) => {
236:                     // NOTE: hash params from `ADMIN_TOKEN` are used instead of what is configured in the `Argon2` instance.
237:                     argon2::Argon2::default().verify_password(token.trim().as_ref(), &h).is_ok()
238:                 }
239:                 Err(e) => {
240:                     error!("The configured Argon2 PHC in `ADMIN_TOKEN` is invalid: {e}");
241:                     false
242:                 }
243:             }
244:         }
245:         Some(t) => crate::crypto::ct_eq(t.trim(), token.trim()),
246:     }
247: }

310:         if CONFIG.mail_enabled() {

646: use cached::macros::cached;
647: /// Cache this function to prevent API call rate limit. Github only allows 60 requests per hour, and we use 3 here already
648: /// It will cache this function for 600 seconds (10 minutes) which should prevent the exhaustion of the rate limit
649: /// Any cache will be lost if Vaultwarden is restarted
650: #[cached(ttl = 600, sync_writes = "default")]
651: async fn get_release_info(has_http_access: bool) -> (String, String, String) {
652:     // If the HTTP Check failed, do not even attempt to check for new versions since we were not able to connect with github.com anyway.
653:     if has_http_access {
654:         (
655:             match get_json_api::<GitRelease>("https://api.github.com/repos/dani-garcia/vaultwarden/releases/latest")
656:                 .await
657:             {
658:                 Ok(r) => r.tag_name,
659:                 _ => "-".to_owned(),
660:             },
661:             match get_json_api::<GitCommit>("https://api.github.com/repos/dani-garcia/vaultwarden/commits/main").await {
662:                 Ok(mut c) => {
663:                     c.sha.truncate(8);
664:                     c.sha
665:                 }
666:                 _ => "-".to_owned(),
667:             },
668:             // Do not fetch the web-vault version when running within a container
669:             // The web-vault version is embedded within the container it self, and should not be updated manually
670:             match get_json_api::<GitRelease>("https://api.github.com/repos/dani-garcia/bw_web_builds/releases/latest")
671:                 .await
672:             {
673:                 Ok(r) => r.tag_name.trim_start_matches('v').to_owned(),
674:                 _ => "-".to_owned(),
675:             },
676:         )
677:     } else {
678:         ("-".to_owned(), "-".to_owned(), "-".to_owned())
679:     }

682: async fn get_ntp_time(has_http_access: bool) -> String {
683:     if has_http_access && let Ok(cf_trace) = get_text_api("https://cloudflare.com/cdn-cgi/trace").await {
684:         for line in cf_trace.lines() {
685:             if let Some((key, value)) = line.split_once('=')
686:                 && key == "ts"
687:             {
688:                 let ts = value.split_once('.').map_or(value, |(s, _)| s);
689:                 if let Ok(dt) = chrono::DateTime::parse_from_str(ts, "%s") {
690:                     return dt.format("%Y-%m-%d %H:%M:%S UTC").to_string();
691:                 }
692:                 break;
693:             }
694:         }
695:     }
696:     String::from("Unable to fetch NTP time.")

699: fn web_vault_compare(active: &str, latest: &str) -> i8 {
700:     use semver::Version;
701:     use std::cmp::Ordering;
702: 
703:     let active_semver = Version::parse(active).unwrap_or_else(|e| {
704:         warn!("Unable to parse active web-vault version '{active}': {e}");
705:         Version::parse("2025.1.1").unwrap()
706:     });
707:     let latest_semver = Version::parse(latest).unwrap_or_else(|e| {
708:         warn!("Unable to parse latest web-vault version '{latest}': {e}");
709:         Version::parse("2025.1.1").unwrap()
710:     });
711: 
712:     match active_semver.cmp(&latest_semver) {
713:         Ordering::Less => -1,
714:         Ordering::Equal => 0,
715:         Ordering::Greater => 1,
716:     }
717: }

735: fn folder_has_hbs_files(dir: &std::path::Path) -> bool {
736:     let Ok(files) = std::fs::read_dir(dir) else {
737:         // No files in this directory at all, so we can return false
738:         return false;
739:     };
740: 
741:     files.flatten().any(|f| {
742:         // Validate if it is a file and if it has the `.hbs` extension and starts with a-z or 0-9
743:         f.file_type().is_ok_and(|t| t.is_file())
744:             && f.path().extension().is_some_and(|e| e.eq_ignore_ascii_case("hbs"))
745:             && f.file_name().to_str().is_some_and(|n| n.starts_with(|c: char| c.is_ascii_alphanumeric()))
746:     })
747: }

861: #[rocket::async_trait]
862: impl<'r> FromRequest<'r> for AdminToken {
863:     type Error = &'static str;
864: 
865:     async fn from_request(request: &'r Request<'_>) -> Outcome<Self, Self::Error> {
866:         let Outcome::Success(ip) = ClientIp::from_request(request).await else {
867:             err_handler!("Error getting Client IP")
868:         };
869: 
870:         if !CONFIG.disable_admin_token() {
871:             let cookies = request.cookies();
872: 
873:             let access_token = if let Some(cookie) = cookies.get(COOKIE_NAME) {
874:                 cookie.value()
875:             } else {
876:                 let requested_page =
877:                     request.segments::<std::path::PathBuf>(0..).unwrap_or_default().display().to_string();
878:                 // When the requested page is empty, it is `/admin`, in that case, Forward, so it will render the login page
879:                 // Else, return a 401 failure, which will be caught
880:                 if requested_page.is_empty() {
881:                     return Outcome::Forward(Status::Unauthorized);
882:                 }
883:                 return Outcome::Error((Status::Unauthorized, "Unauthorized"));
884:             };
885: 
886:             if decode_admin(access_token).is_err() {
887:                 // Remove admin cookie
888:                 cookies.remove(Cookie::build(COOKIE_NAME).path(admin_path()));
889:                 error!("Invalid or expired admin JWT. IP: {}.", ip.ip);
890:                 return Outcome::Error((Status::Unauthorized, "Session expired"));
891:             }
892:         }
893: 
894:         Outcome::Success(Self {
895:             ip,
896:         })
897:     }
898: }
